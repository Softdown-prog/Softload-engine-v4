// -----------------------------------------------------------------
// Busca Instantânea em Tempo Real (Blogger JSON Feed)
// -----------------------------------------------------------------
const searchInput = document.getElementById('search-input');
const searchResults = document.getElementById('search-results');


if (searchInput && searchResults) {
  let debounceTimer;


  searchInput.addEventListener('input', (e) => {
    clearTimeout(debounceTimer);
    const query = e.target.value.trim();


    if (query.length < 3) {
      searchResults.innerHTML = '';
      searchResults.classList.add('d-none');
      return;
    }


    debounceTimer = setTimeout(async () => {
      try {
        const response = await fetch(`/feeds/posts/summary?alt=json&q=${encodeURIComponent(query)}&max-results=5`);
        const data = await response.json();
        const entries = data.feed.entry || [];


        if (entries.length === 0) {
          searchResults.innerHTML = '<div class="p-3 text-muted small text-center">Nenhum resultado encontrado.</div>';
        } else {
          searchResults.innerHTML = entries.map(entry => {
            const title = entry.title.$t;
            const link = entry.link.find(l => l.rel === 'alternate').href;
            const thumb = entry.media$thumbnail ? entry.media$thumbnail.url.replace('/s72-c/', '/w100-h100-c/') : 'https://via.placeholder.com/50';


            return `
              <a href="${link}" class="list-group-item list-group-item-action d-flex align-items-center gap-2 p-2">
                <img src="${thumb}" width="40" height="40" class="rounded object-fit-cover" alt=""/>
                <span class="small fw-semibold text-truncate">${title}</span>
              </a>
            `;
          }).join('');
        }
        searchResults.classList.remove('d-none');
      } catch (err) {
        console.error('Erro na busca instantânea:', err);
      }
    }, 300);
  });


  // Fechar resultados ao clicar fora
  document.addEventListener('click', (e) => {
    if (!searchInput.contains(e.target) && !searchResults.contains(e.target)) {
      searchResults.classList.add('d-none');
    }
  });
}

Precarregamento ao Passar o Mouse (Instant Click / Instant Page)
// -----------------------------------------------------------------
// Prefetch Inteligente ao Focar/Hover em Links Internos
// -----------------------------------------------------------------
document.querySelectorAll('a[href^="/"]').forEach(link => {
  link.addEventListener('mouseenter', () => {
    const href = link.getAttribute('href');
    if (href && !document.querySelector(`link[rel="prefetch"][href="${href}"]`)) {
      const prefetchLink = document.createElement('link');
      prefetchLink.rel = 'prefetch';
      prefetchLink.href = href;
      document.head.appendChild(prefetchLink);
    }
  }, { once: true });
});

// -----------------------------------------------------------------
// Otimizador Dinâmico de Imagens do Blogger
// -----------------------------------------------------------------
document.querySelectorAll('.post-body img, .post-thumb-img').forEach(img => {
  let src = img.getAttribute('src');
  if (src && src.includes('bp.blogspot.com')) {
    // Redimensiona proporcionalmente para 800px de largura com compressão WebP (/w800-rw/)
    src = src.replace(/\/s\d+(-c)?\//, '/w800-rw/').replace(/\/w\d+-h\d+(-c)?\//, '/w800-rw/');
    img.setAttribute('src', src);
  }
  img.setAttribute('loading', 'lazy');
  img.setAttribute('decoding', 'async');
});

Como replicar para novos ícones no futuro:
Para criar novos links com ícones SVG sem carregar fontes externas pesadas (como FontAwesome):
 * Mantenha o atributo viewBox="0 0 24 24" nos SVGs.
 * Certifique-se de que o elemento pai (.nav-link, .dropdown-item ou .badge-category) já tem o CSS do exemplo aplicado, garantindo o alinhamento e o dimensionamento perfeitos.




Para deixar seu template 100% completo, o que faltava era o tratamento CSS dos links dentro do conteúdo dos artigos (.post-body a) e a estilização fina dos elementos internos da postagem (como citações, tabelas, imagens e barra de rolagem).
